// core_crypto/src/lib.rs
pub fn encrypt_voice_frame(
    frame: &[u8],
    key: &SessionKey,
    nonce: &XNonce,
) -> Result<Vec<u8>> {
    let cipher = XChaCha20Poly1305::new(key);
    cipher.encrypt(nonce, frame)
        .map_err(|_| CryptoError::EncryptionFailed)
}
